47 bool listDir(const char* path,
48 std::function<bool(const FileInfo&)> callback) override {
49 // Fast path for VFS mounted file systems (e.g. SD_MMC, SD, FFat): use
50 // the POSIX opendir()/readdir()/stat() API directly instead of the
51 // Arduino fs::File API. This avoids constructing/destructing a File
52 // object (and its extra Stream/Print indirection) for every entry,
53 // which otherwise dominates the time for directories with many files
54 // or subdirectories.
55 const char* mp = fs.mountpoint();
56 if (mp != nullptr) {
57 std::string full = std::string(mp) + path;
58 DIR* d = opendir(full.c_str());
59 if (d == nullptr) return false;
60 FileInfo info;
61 struct dirent* entry;
62 bool more = true;
63 while (more && (entry = readdir(d)) != nullptr) {
64 if (strcmp(entry->d_name, ".") == 0 ||
65 strcmp(entry->d_name, "..") == 0)
66 continue;
67 info.name = entry->d_name;
68 std::string childPath = full == "/" ? full + info.name
69 : full + "/" + info.name;
70 struct stat st;
71 if (::stat(childPath.c_str(), &st) == 0) {
72 info.isDirectory = S_ISDIR(st.st_mode);
73 info.size = info.isDirectory ? 0 : (uint64_t)st.st_size;
74 info.modified = toUnix(st.st_mtime);
75 } else {
76#ifdef DT_DIR
77 info.isDirectory = entry->d_type == DT_DIR;
78#else
79 info.isDirectory = false;
80#endif
81 info.size = 0;
82 info.modified = 0;
83 }
84 more = callback(info);
85 }
86 closedir(d);
87 return true;
88 }
89 // fallback: Arduino fs::File based listing (e.g. LittleFS, SPIFFS)
90 fs::File dir = fs.open(path, FILE_READ);
91 if (!dir || !dir.isDirectory()) return false;
92 FileInfo info;
93 for (fs::File f = dir.openNextFile(); f; f = dir.openNextFile()) {
94 info.name = baseName(f.name());
95 info.isDirectory = f.isDirectory();
96 info.size = info.isDirectory ? 0 : f.size();
97 info.modified = toUnix(f.getLastWrite());
98 f.close();
99 if (!callback(info)) break;
100 }
101 dir.close();
102 return true;
103 }

113 int64_t write(const char* path, uint64_t offset, const uint8_t* buffer,
114 size_t len) override {
115 fs::File* f = openFile(path, true);
116 if (f == nullptr) return -1;
117 if (f->position() != offset) {
118 size_t size = f->size();
119 if (offset > size) {
120 // the file must grow: fill the gap with zeros
121 if (!f->seek(size)) return -1;
122 uint8_t zeros[512] = {0};
123 for (uint64_t pos = size; pos < offset;) {
124 size_t n = offset - pos > sizeof(zeros) ? sizeof(zeros)
125 : (size_t)(offset - pos);
126 if (f->write(zeros, n) != n) return -1;
127 pos += n;
128 }
129 } else if (!f->seek(offset)) {
130 return -1;
131 }
132 }
133 return f->write(buffer, len);
134 }

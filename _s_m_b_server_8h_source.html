350 bool processConnection(Connection& c) {
351 if (!c.client.connected()) return false;
352 int avail = c.client.available();
353 while (avail > 0) {
354 size_t old = c.rx.size();
355 if (old == 0) c.rxStartUs = micros();
356 size_t n = avail > 4096 ? 4096 : (size_t)avail;
357 c.rx.resize(old + n);
358#if defined(IS_DESKTOP)
359 int rc = c.client.readBytes(c.rx.data() + old, n);
360#else
361 int rc = c.client.read(c.rx.data() + old, n);
362#endif
363 c.rx.resize(old + (rc > 0 ? rc : 0));
364 if (rc <= 0) break;
365 // process all complete frames
366 while (c.rx.size() >= 4) {
367 size_t len = ((size_t)c.rx[1] << 16) | ((size_t)c.rx[2] << 8) |
368 (size_t)c.rx[3];
369 if (len > maxIO + 4096) {
370 SMB_LOGE("message too big: %u", (unsigned)len);
371 return false;
372 }
373 if (c.rx.size() < len + 4) break;
374 c.receiveUs = micros() - c.rxStartUs;
375 if (!processFrame(c, c.rx[0], c.rx.data() + 4, len)) return false;
376 c.rx.erase(c.rx.begin(), c.rx.begin() + len + 4);
377 c.rxStartUs = micros(); // the next request is already arriving
378 }
379 avail = c.client.available();
380 }
381 if (c.rx.empty() && c.rx.capacity() > 8192) {
382 PsramVector<uint8_t>().swap(c.rx);
383 }
384 return true;
385 }

387 bool processFrame(Connection& c, uint8_t type, const uint8_t* data,
388 size_t len) {
389 if (type == 0x85) return true; // keep alive
390 if (type == 0x81) {
391 // NetBIOS session request (port 139): positive response
392 const uint8_t ok[4] = {0x82, 0, 0, 0};
393 return send(c, ok, 4);
394 }
395 if (type != 0 || len < 4) return false;
396 if (data[0] == 0xFF && memcmp(data + 1, "SMB", 3) == 0) {
397 return processSmb1(c, data, len);
398 }
399 if (data[0] == 0xFE && memcmp(data + 1, "SMB", 3) == 0) {
400 return processSmb2(c, data, len);
401 }
402 SMB_LOGE("unsupported protocol 0x%02x", data[0]);
403 return false;
404 }

406 bool send(Connection& c, const uint8_t* data, size_t len) {
407 size_t sent = 0;
408 uint32_t start = millis();
409 while (sent < len) {
410 size_t n = c.client.write(data + sent, len - sent);
411 // some implementations return (size_t)-1 if the socket would block
412 if (n == 0 || n > len - sent) {
413 if (!c.client.connected() || millis() - start > 10000) return false;
414 delay(1);
415 continue;
416 }
417 sent += n;
418 }
419 return true;
420 }

423 bool processSmb1(Connection& c, const uint8_t* data, size_t len) {
424 if (len < 35 || data[4] != 0x72) return false;
425 std::string dialects((const char*)data + 35, len - 35);
426 uint16_t dialect = 0;
427 if (dialects.find("SMB 2.???") != std::string::npos) {
428 dialect = SMB2_DIALECT_WILDCARD;
429 } else if (dialects.find("SMB 2.002") != std::string::npos) {
430 dialect = SMB2_DIALECT_0202;
431 }
432 if (dialect == 0) {
433 SMB_LOGI("client does not support SMB2");
434 return false;
435 }
436 if (dialect != SMB2_DIALECT_WILDCARD) c.dialect = dialect;
437 resp.assign(SMB2_HEADER_SIZE, 0);
438 Writer w(resp);
439 writeNegotiateBody(w, dialect);
440 writeHeader(resp, 0, STATUS_SUCCESS, SMB2_NEGOTIATE, 1, 0, 0, 0, 0);
441 return sendResponses(c, {resp});
442 }

444 bool sendResponses(Connection& c, const std::vector<PsramVector<uint8_t>>& r) {
445 tx.assign(4, 0);
446 for (auto& part : r) tx.insert(tx.end(), part.begin(), part.end());
447 size_t len = tx.size() - 4;
448 tx[1] = (uint8_t)(len >> 16);
449 tx[2] = (uint8_t)(len >> 8);
450 tx[3] = (uint8_t)len;
451 bool ok = send(c, tx.data(), tx.size());
452 if (tx.capacity() > 16384) PsramVector<uint8_t>().swap(tx);
453 return ok;
454 }

458 bool processSmb2(Connection& c, const uint8_t* data, size_t len) {
459 uint32_t startUs = micros();
460 c.timedOpen = 0;
461 c.storageUs = 0;
462 std::vector<PsramVector<uint8_t>> responses;
463 std::vector<SignInfo> signInfos;
464 Request req;
465 size_t off = 0;
466 while (off + SMB2_HEADER_SIZE <= len) {
467 uint32_t next = Reader(data + off, len - off).u32(20);
468 size_t msgLen = next ? next : len - off;
469 if (msgLen < SMB2_HEADER_SIZE || off + msgLen > len) return false;
470 const uint8_t* msg = data + off;
471 req.r = Reader(msg, msgLen);
472 req.command = req.r.u16(12);
473 req.flags = req.r.u32(16);
474 uint64_t messageId = req.r.u64(24);
475 uint16_t credits = req.r.u16(14);
476 uint16_t charge = req.r.u16(6);
477 if (!req.related()) {
478 req.sessionId = req.r.u64(40);
479 req.treeId = req.r.u32(36);
480 req.relatedFileId = ~0ULL;
481 req.relatedStatus = STATUS_SUCCESS;
482 } else {
483 if (req.r.u64(40) != ~0ULL) req.sessionId = req.r.u64(40);
484 if (req.r.u32(36) != ~0U) req.treeId = req.r.u32(36);
485 }
486 if (req.command == SMB2_CANCEL) {
487 off += msgLen;
488 if (!next) break;
489 continue; // no response
490 }
491
492 // capture the signing key: the session might be removed by LOGOFF
493 SignInfo sign;
494 Session* before = findSession(c, req.sessionId);
495 if (before != nullptr) sign = SignInfo(*before);
496
497 resp.assign(SMB2_HEADER_SIZE, 0);
498 keepErrorBody = false;
499 uint32_t status = dispatch(c, req, msg, msgLen);
500 if (status != STATUS_SUCCESS && status != STATUS_BUFFER_OVERFLOW &&
501 status != STATUS_MORE_PROCESSING_REQUIRED && !keepErrorBody) {
502 writeError();
503 }
504 if (req.command == SMB2_CREATE && status != STATUS_SUCCESS) {
505 req.relatedStatus = status;
506 }
507 SMB_LOGD("cmd %u -> 0x%08x", req.command, (unsigned)status);
508 Session* s = findSession(c, req.sessionId);
509 if (s != nullptr) sign = SignInfo(*s);
510 writeHeader(resp, req.r.u32(32), status, req.command,
511 credits ? credits : 1, charge, messageId, req.treeId,
512 req.sessionId, req.flags & SMB2_FLAGS_RELATED_OPERATIONS);
513 sign.active = sign.hasKey && (sign.required ||
514 (req.flags & SMB2_FLAGS_SIGNED));
515 if (req.command == SMB2_SESSION_SETUP && status != STATUS_SUCCESS)
516 sign.active = false;
517 responses.push_back(resp);
518 signInfos.push_back(sign);
519 off += msgLen;
520 if (!next) break;
521 }
522 if (responses.empty()) return true;
523 // link compound responses and sign them
524 for (size_t i = 0; i < responses.size(); i++) {
525 auto& r = responses[i];
526 bool last = i + 1 == responses.size();
527 if (!last) {
528 while (r.size() % 8) r.push_back(0);
529 Writer(r).put32(20, (uint32_t)r.size());
530 }
531 if (signInfos[i].active) signMessage(r.data(), r.size(), signInfos[i].key);
532 }
533 if (c.timedOpen == 0) return sendResponses(c, responses);
534 uint32_t sendStartUs = micros();
535 bool ok = sendResponses(c, responses);
536 Open* o = findOpenById(c, c.timedOpen);
537 if (o != nullptr) {
538 Timing& t = o->timing;
539 t.storageUs += c.storageUs;
540 t.cpuUs += sendStartUs - startUs - c.storageUs;
541 t.sendUs += micros() - sendStartUs;
542 t.receiveUs += c.receiveUs;
543 }
544 return ok;
545 }

582 void logTiming(const Open& o, const char* label, uint32_t closeUs) {
583 const Timing& t = o.timing;
584 uint64_t bytes = t.bytesRead + t.bytesWritten;
585 uint32_t ms = millis() - t.startMs;
586 uint64_t storage = t.storageUs + closeUs;
587 uint64_t wallUs = (uint64_t)ms * 1000;
588 uint64_t known = storage + t.cpuUs + t.sendUs + t.receiveUs;
589 uint64_t wait = wallUs > known ? wallUs - known : 0;
590 auto pct = [&](uint64_t us) {
591 return wallUs ? (unsigned)(us * 100 / wallUs) : 0u;
592 };
593 SMB_LOGI("timing %s %s: %u reads, %u writes, %llu bytes in %u ms (%u "
594 "KB/s)",
595 label, o.path.c_str(), (unsigned)t.reads, (unsigned)t.writes,
596 (unsigned long long)bytes, (unsigned)ms,
597 ms ? (unsigned)(bytes / ms) : 0u);
598 SMB_LOGI(
599 "timing storage %u ms (%u%%), cpu %u ms (%u%%), send %u ms (%u%%), "
600 "receive %u ms (%u%%), wait %u ms (%u%%)",
601 (unsigned)(storage / 1000), pct(storage), (unsigned)(t.cpuUs / 1000),
602 pct(t.cpuUs), (unsigned)(t.sendUs / 1000), pct(t.sendUs),
603 (unsigned)(t.receiveUs / 1000), pct(t.receiveUs),
604 (unsigned)(wait / 1000), pct(wait));
605 }

607 uint32_t dispatch(Connection& c, Request& req, const uint8_t* msg,
608 size_t len) {
609 req.session = nullptr;
610 req.tree = nullptr;
611 switch (req.command) {
612 case SMB2_NEGOTIATE:
613 return onNegotiate(c, req);
614 case SMB2_SESSION_SETUP:
615 return onSessionSetup(c, req);
616 case SMB2_ECHO:
617 return body16(4);
618 default:
619 break;
620 }
621 // all other commands require a valid session
622 req.session = findSession(c, req.sessionId);
623 if (req.session == nullptr || !req.session->authenticated)
624 return STATUS_USER_SESSION_DELETED;
625 if ((req.flags & SMB2_FLAGS_SIGNED) && req.session->hasKey &&
626 !verifySignature(msg, len, req.session->key)) {
627 SMB_LOGI("invalid signature");
628 return STATUS_ACCESS_DENIED;
629 }
630 if (req.command == SMB2_LOGOFF) return onLogoff(c, req);
631 if (req.command == SMB2_TREE_CONNECT) return onTreeConnect(c, req);
632
633 // all other commands require a valid tree
634 req.tree = findTree(c, req.treeId, req.sessionId);
635 if (req.tree == nullptr) return STATUS_NETWORK_NAME_DELETED;
636 switch (req.command) {
637 case SMB2_TREE_DISCONNECT:
638 return onTreeDisconnect(c, req);
639 case SMB2_CREATE:
640 return onCreate(c, req);
641 case SMB2_CLOSE:
642 return onClose(c, req);
643 case SMB2_FLUSH:
644 return onFlush(c, req);
645 case SMB2_READ:
646 return onRead(c, req);
647 case SMB2_WRITE:
648 return onWrite(c, req);
649 case SMB2_LOCK:
650 return findOpen(c, req, 72) ? body16(4) : STATUS_FILE_CLOSED;
651 case SMB2_IOCTL:
652 return onIoctl(c, req);
653 case SMB2_QUERY_DIRECTORY:
654 return onQueryDirectory(c, req);
655 case SMB2_QUERY_INFO:
656 return onQueryInfo(c, req);
657 case SMB2_SET_INFO:
658 return onSetInfo(c, req);
659 case SMB2_CHANGE_NOTIFY:
660 default:
661 return STATUS_NOT_SUPPORTED;
662 }

668 uint16_t command, uint16_t credits, uint16_t charge,

684 w.put32(32, pid);
685 w.put32(36, treeId);

707 static void signMessage(uint8_t* msg, size_t len, const uint8_t key[16]) {
708 uint32_t flags = Reader(msg, len).u32(16) | SMB2_FLAGS_SIGNED;
709 msg[16] = (uint8_t)flags;
710 msg[17] = (uint8_t)(flags >> 8);
711 msg[18] = (uint8_t)(flags >> 16);
712 msg[19] = (uint8_t)(flags >> 24);
713 memset(msg + 48, 0, 16);
714 uint8_t mac[32];
715 Crypto::hmacSha256(key, 16, msg, len, nullptr, 0, mac);
716 memcpy(msg + 48, mac, 16);
717 }
718
719 static bool verifySignature(const uint8_t* msg, size_t len,
720 const uint8_t key[16]) {
721 uint8_t mac[32];
722 // the signature is calculated with a zeroed signature field
723 uint8_t header[SMB2_HEADER_SIZE];
724 memcpy(header, msg, SMB2_HEADER_SIZE);
725 memset(header + 48, 0, 16);
726 Crypto::hmacSha256(key, 16, header, SMB2_HEADER_SIZE,
727 msg + SMB2_HEADER_SIZE, len - SMB2_HEADER_SIZE, mac);
728 return Crypto::equals(mac, msg + 48, 16);
729 }

734 void writeNegotiateBody(Writer<Vec>& w, uint16_t dialect) {
735 std::vector<uint8_t> token = NtlmAuth::negotiateToken();
736 w.u16(65);
737 w.u16(SMB2_NEGOTIATE_SIGNING_ENABLED |
738 (signingRequired ? SMB2_NEGOTIATE_SIGNING_REQUIRED : 0));
739 w.u16(dialect);
740 w.u16(0);
741 w.bytes(serverGuid, 16);
742 w.u32(0); // capabilities
743 w.u32(maxIO);
744 w.u32(maxIO);
745 w.u32(maxIO);
746 w.u64(toFileTime(Platform::unixTime()));
747 w.u64(startTime);
748 w.u16(SMB2_HEADER_SIZE + 64);
749 w.u16((uint16_t)token.size());
750 w.u32(0);
751 w.bytes(token.data(), token.size());
752 }

754 uint32_t onNegotiate(Connection& c, Request& req) {
755 uint16_t count = req.r.u16(66);
756 uint16_t best = 0;
757 for (int i = 0; i < count; i++) {
758 uint16_t d = req.r.u16(100 + 2 * i);
759 if ((d == SMB2_DIALECT_0202 || d == SMB2_DIALECT_0210) && d > best)
760 best = d;
761 }
762 if (best == 0) {
763 SMB_LOGI("no supported dialect");
764 return STATUS_NOT_SUPPORTED;
765 }
766 c.dialect = best;
767 Writer w(resp);
768 writeNegotiateBody(w, best);
769 return STATUS_SUCCESS;
770 }

788 uint32_t onSessionSetup(Connection& c, Request& req) {
789 Session* s = nullptr;
790 if (req.sessionId == 0) {
791 c.sessions.emplace_back();
792 s = &c.sessions.back();
793 s->id = nextSessionId++;
794 s->auth.setNames(serverName, domainName);
795 req.sessionId = s->id;
796 } else {
797 s = findSession(c, req.sessionId);
798 if (s == nullptr) return STATUS_USER_SESSION_DELETED;
799 }
800 uint8_t clientMode = req.r.u8(67);
801 uint16_t blobOff = req.r.u16(76);
802 uint16_t blobLen = req.r.u16(78);
803 const uint8_t* blob = req.r.ptr(blobOff, blobLen);
804 if (blob == nullptr) return STATUS_INVALID_PARAMETER;
805
806 std::vector<uint8_t> out;
807 AuthResult result = s->auth.process(blob, blobLen, users, guestAccess, out);
808 uint16_t sessionFlags = 0;
809 uint32_t status = STATUS_SUCCESS;
810 switch (result) {
811 case AuthResult::Continue:
812 status = STATUS_MORE_PROCESSING_REQUIRED;
813 break;
814 case AuthResult::Success:
815 s->authenticated = true;
816 s->readOnly = s->auth.isReadOnly();
817 s->hasKey = true;
818 memcpy(s->key, s->auth.sessionKey(), 16);
819 s->signingRequired =
820 signingRequired || (clientMode & SMB2_NEGOTIATE_SIGNING_REQUIRED);
821 SMB_LOGI("user '%s' logged in%s", s->auth.userName().c_str(),
822 s->readOnly ? " (read-only)" : "");
823 break;
824 case AuthResult::Guest:
825 s->authenticated = true;
826 s->guest = true;
827 sessionFlags = SMB2_SESSION_FLAG_IS_GUEST;
828 SMB_LOGI("guest logged in");
829 break;
830 default:
831 SMB_LOGI("login failed for '%s'", s->auth.userName().c_str());
832 removeSession(c, s->id);
833 return STATUS_LOGON_FAILURE;
834 }
835 Writer w(resp);
836 w.u16(9);
837 w.u16(sessionFlags);
838 w.u16(out.empty() ? 0 : SMB2_HEADER_SIZE + 8);
839 w.u16((uint16_t)out.size());
840 w.bytes(out.data(), out.size());
841 return status;
842 }

957 static bool toPath(std::string name, std::string& path) {
958 for (auto& ch : name)
959 if (ch == '\\') ch = '/';
960 // strip default data stream suffix
961 const char* streams[] = {"::$DATA", ":$DATA"};
962 for (auto s : streams) {
963 size_t n = strlen(s);
964 if (name.size() >= n && name.compare(name.size() - n, n, s) == 0)
965 name.resize(name.size() - n);
966 }
967 if (name.find_first_of(":*?<>\"|") != std::string::npos) return false;
968 path = "/";
969 size_t start = 0;
970 while (start <= name.size()) {
971 size_t end = name.find('/', start);
972 if (end == std::string::npos) end = name.size();
973 std::string part = name.substr(start, end - start);
974 if (part == "..") return false;
975 if (!part.empty() && part != ".") {
976 if (path.size() > 1) path += "/";
977 path += part;
978 }
979 start = end + 1;
980 }
981 return true;
982 }

1036 uint32_t onCreate(Connection& c, Request& req) {
1037 uint32_t disposition = req.r.u32(100);
1038 uint32_t options = req.r.u32(104);
1039 std::string name = req.r.utf16(req.r.u16(108), req.r.u16(110));
1040 if (req.r.hasError()) return STATUS_INVALID_PARAMETER;
1041 if (c.opens.size() >= 256) return STATUS_INSUFFICIENT_RESOURCES;
1042
1043 Open open;
1044 open.treeId = req.tree->id;
1045 open.sessionId = req.sessionId;
1046 FileInfo info;
1047 uint32_t action = FILE_OPENED;
1048
1049 Share* share = shareOf(req);
1050 if (share == nullptr) {
1051 // IPC$: named pipes
1052 std::string pipe = name;
1053 for (auto& ch : pipe) ch = (char)tolower((unsigned char)ch);
1054 if (pipe.rfind("\\pipe\\", 0) == 0) pipe = pipe.substr(6);
1055 if (pipe != "srvsvc") return STATUS_OBJECT_NAME_NOT_FOUND;
1056 open.isPipe = true;
1057 open.path = pipe;
1058 info.name = pipe;
1059 } else {
1060 if (!toPath(name, open.path)) return STATUS_OBJECT_NAME_INVALID;
1061 FileSystem* fs = share->fs;
1062 std::string path = fsPath(*share, open.path);
1063 bool exists = fs->stat(path.c_str(), info);
1064 if (open.path == "/") {
1065 exists = true;
1066 info.isDirectory = true;
1067 }
1068 bool isDirRequest = options & FILE_DIRECTORY_FILE;
1069 if (!exists) {
1070 if (disposition == FILE_OPEN || disposition == FILE_OVERWRITE) {
1071 return parentExists(*share, open.path) ? STATUS_OBJECT_NAME_NOT_FOUND
1072 : STATUS_OBJECT_PATH_NOT_FOUND;
1073 }
1074 if (req.tree->readOnly) return STATUS_MEDIA_WRITE_PROTECTED;
1075 if (!parentExists(*share, open.path))
1076 return STATUS_OBJECT_PATH_NOT_FOUND;
1077 bool ok = isDirRequest ? fs->mkdir(path.c_str())
1078 : fs->createFile(path.c_str());
1079 if (!ok || !fs->stat(path.c_str(), info)) return STATUS_ACCESS_DENIED;
1080 action = FILE_CREATED;
1081 } else {
1082 if (disposition == FILE_CREATE) return STATUS_OBJECT_NAME_COLLISION;
1083 if (isDirRequest && !info.isDirectory) return STATUS_NOT_A_DIRECTORY;
1084 if ((options & FILE_NON_DIRECTORY_FILE) && info.isDirectory)
1085 return STATUS_FILE_IS_A_DIRECTORY;
1086 if (disposition == FILE_SUPERSEDE || disposition == FILE_OVERWRITE ||
1087 disposition == FILE_OVERWRITE_IF) {
1088 if (info.isDirectory) return STATUS_INVALID_PARAMETER;
1089 if (req.tree->readOnly) return STATUS_MEDIA_WRITE_PROTECTED;
1090 if (!fs->createFile(path.c_str())) return STATUS_ACCESS_DENIED;
1091 info.size = 0;
1092 action = disposition == FILE_SUPERSEDE ? FILE_SUPERSEDED
1093 : FILE_OVERWRITTEN;
1094 }
1095 }
1096 open.isDir = info.isDirectory;
1097 if (options & FILE_DELETE_ON_CLOSE) {
1098 if (req.tree->readOnly) return STATUS_MEDIA_WRITE_PROTECTED;
1099 if (open.isDir && !isEmptyDir(*fs, path))
1100 return STATUS_DIRECTORY_NOT_EMPTY;
1101 open.deleteOnClose = true;
1102 }
1103 }
1104 open.id = nextFileId++;
1105 c.opens.push_back(open);
1106 req.relatedFileId = open.id;
1107 req.relatedStatus = STATUS_SUCCESS;
1108
1109 uint64_t t = toFileTime(info.modified);
1110 Writer w(resp);
1111 w.u16(89);
1112 w.u8(0); // no oplock
1113 w.u8(0);
1114 w.u32(action);
1115 w.u64(t);
1116 w.u64(t);
1117 w.u64(t);
1118 w.u64(t);
1119 w.u64(allocationSize(info.size));
1120 w.u64(info.size);
1121 w.u32(open.isPipe ? FILE_ATTRIBUTE_NORMAL : attributes(info));
1122 w.u32(0);
1123 w.u64(open.id);
1124 w.u64(open.id);
1125 w.u32(0); // no create contexts
1126 w.u32(0);
1127 w.u8(0);
1128 SMB_LOGD("open %s", open.path.c_str());
1129 return STATUS_SUCCESS;
1130 }

1150 void closeOpen(Connection& c, size_t idx) {
1151 Open o = c.opens[idx];
1152 c.opens.erase(c.opens.begin() + idx);
1153 if (o.isPipe) return;
1154 Tree* tree = nullptr;
1155 for (auto& t : c.trees)
1156 if (t.id == o.treeId) tree = &t;
1157 if (tree == nullptr || tree->share < 0) return;
1158 Share& share = shares[tree->share];
1159 std::string path = fsPath(share, o.path);
1160 bool stillOpen = false;
1161 for (auto& other : c.opens)
1162 if (other.treeId == o.treeId && other.path == o.path) stillOpen = true;
1163 uint32_t closeStartUs = micros();
1164 if (!stillOpen) share.fs->close(path.c_str());
1165 // closing flushes the written data
1166 if (timingLog && (o.timing.reads || o.timing.writes))
1167 logTiming(o, "done", micros() - closeStartUs);
1168 if (o.deleteOnClose) {
1169 bool ok = o.isDir ? share.fs->rmdir(path.c_str())
1170 : share.fs->remove(path.c_str());
1171 SMB_LOGI("delete %s: %s", path.c_str(), ok ? "ok" : "failed");
1172 }

1210 uint32_t onRead(Connection& c, Request& req) {
1211 uint32_t length = req.r.u32(68);
1212 uint64_t offset = req.r.u64(72);
1213 Open* o = findOpen(c, req, 80);
1214 if (o == nullptr) return fileStatus(req);
1215 if (o->isDir) return STATUS_INVALID_DEVICE_REQUEST;
1216 if (length > maxIO) length = maxIO;
1217
1218 const size_t dataOff = SMB2_HEADER_SIZE + 16;
1219 resp.resize(dataOff);
1220 uint32_t status = STATUS_SUCCESS;
1221 int64_t n;
1222 if (o->isPipe) {
1223 n = o->pipeOut.size() < length ? o->pipeOut.size() : length;
1224 resp.insert(resp.end(), o->pipeOut.begin(), o->pipeOut.begin() + n);
1225 o->pipeOut.erase(o->pipeOut.begin(), o->pipeOut.begin() + n);
1226 if (!o->pipeOut.empty()) status = STATUS_BUFFER_OVERFLOW;
1227 } else {
1228 Share* share = shareOf(req);
1229 resp.resize(dataOff + length);
1230 uint32_t startUs = micros();
1231 n = share->fs->read(fsPath(*share, o->path).c_str(), offset,
1232 resp.data() + dataOff, length);
1233 if (timingLog) timeStorage(c, *o, startUs, true, n);
1234 if (n < 0) {
1235 resp.resize(SMB2_HEADER_SIZE);
1236 return STATUS_ACCESS_DENIED;
1237 }
1238 resp.resize(dataOff + n);
1239 if (n == 0 && length > 0) {
1240 resp.resize(SMB2_HEADER_SIZE);
1241 return STATUS_END_OF_FILE;
1242 }
1243 }
1244 Writer w(resp);
1245 w.put16(SMB2_HEADER_SIZE, 17);
1246 resp[SMB2_HEADER_SIZE + 2] = (uint8_t)dataOff;
1247 resp[SMB2_HEADER_SIZE + 3] = 0;
1248 w.put32(SMB2_HEADER_SIZE + 4, (uint32_t)n);
1249 w.put32(SMB2_HEADER_SIZE + 8, 0);
1250 w.put32(SMB2_HEADER_SIZE + 12, 0);
1251 if (n == 0) w.u8(0);
1252 return status;
1253 }

1331 if (a.size() != b.size()) return false;
1332 for (size_t i = 0; i < a.size(); i++)
1333 if (tolower((unsigned char)a[i]) != tolower((unsigned char)b[i]))
1334 return false;
1335 return true;

1339 static bool match(const char* p, const char* s) {
1340 while (*p) {
1341 char pc = *p;
1342 if (pc == '*' || pc == '<') {
1343 while (*p == '*' || *p == '<') p++;
1344 if (!*p) return true;
1345 for (; *s; s++)
1346 if (match(p, s)) return true;
1347 return match(p, s);
1348 }
1349 if (!*s) {
1350 // DOS_DOT and DOS_QM match the end of the name
1351 if (pc == '"' || pc == '>') {
1352 p++;
1353 continue;
1354 }
1355 return false;
1356 }
1357 if (pc != '?' && pc != '>' &&
1358 !(pc == '"' && *s == '.') &&
1359 tolower((unsigned char)pc) != tolower((unsigned char)*s))
1360 return false;
1361 p++;
1362 s++;
1363 }
1364 return *s == 0;
1365 }

1399 if (infoClass == FileIdFullDirectoryInformation) w.u32(0);

1409 uint32_t onQueryDirectory(Connection& c, Request& req) {
1410 uint8_t infoClass = req.r.u8(66);
1411 uint8_t flags = req.r.u8(67);
1412 Open* o = findOpen(c, req, 72);
1413 if (o == nullptr) return fileStatus(req);
1414 std::string pattern = req.r.utf16(req.r.u16(88), req.r.u16(90));
1415 uint32_t maxOut = req.r.u32(92);
1416 if (maxOut > maxIO) maxOut = maxIO;
1417 Share* share = shareOf(req);
1418 if (!o->isDir || share == nullptr) return STATUS_INVALID_PARAMETER;
1419 switch (infoClass) {
1420 case FileDirectoryInformation:
1421 case FileFullDirectoryInformation:
1422 case FileBothDirectoryInformation:
1423 case FileNamesInformation:
1424 case FileIdBothDirectoryInformation:
1425 case FileIdFullDirectoryInformation:
1426 break;
1427 default:
1428 return STATUS_INVALID_INFO_CLASS;
1429 }
1430 if (pattern.empty()) pattern = "*";
1431 if (pattern == "*.*") pattern = "*";
1432 bool restart = (flags & (SMB2_RESTART_SCANS | SMB2_REOPEN)) ||
1433 o->pattern.empty();
1434 if (restart) {
1435 o->dirIndex = 0;
1436 o->pattern = pattern;
1437 o->dirListed = false;
1438 o->dirEntries.clear();
1439 }
1440 bool first = o->dirIndex == 0;
1441 bool single = flags & SMB2_RETURN_SINGLE_ENTRY;
1442
1443 // Read the directory from storage only once per open and cache the
1444 // (pattern-matched) entries; subsequent QUERY_DIRECTORY requests for
1445 // the same open just page through the cache. Without this, clients
1446 // that page large directories in several requests would cause an
1447 // O(n^2) re-scan of storage (very slow on SD/FatFs).
1448 std::string path = fsPath(*share, o->path);
1449 if (!o->dirListed) {
1450 o->dirListed = true;
1451 if (!hasWildcards(o->pattern)) {
1452 // lookup of a single name
1453 FileInfo info;
1454 std::string p = (path == "/" ? "" : path) + "/" + o->pattern;
1455 if (share->fs->stat(p.c_str(), info)) {
1456 info.name = o->pattern;
1457 o->dirEntries.push_back(info);
1458 }
1459 } else {
1460 FileInfo dot;
1461 dot.isDirectory = true;
1462 dot.name = ".";
1463 o->dirEntries.push_back(dot);
1464 dot.name = "..";
1465 o->dirEntries.push_back(dot);
1466 share->fs->listDir(path.c_str(), [&](const FileInfo& info) {
1467 if (match(o->pattern.c_str(), info.name.c_str()))
1468 o->dirEntries.push_back(info);
1469 return true;
1470 });
1471 }
1472 }
1473
1474 Writer w(resp);
1475 w.u16(9);
1476 w.u16(SMB2_HEADER_SIZE + 8);
1477 w.u32(0); // length: patched
1478 const size_t start = resp.size();
1479 size_t lastEntry = 0;
1480 int added = 0;
1481 bool full = false;
1482 std::string dirPath = o->path;
1483
1484 while (o->dirIndex < o->dirEntries.size()) {
1485 const FileInfo& info = o->dirEntries[o->dirIndex];
1486 size_t before = resp.size();
1487 if (added > 0) w.align(8, start);
1488 size_t entry = resp.size();
1489 writeDirEntry(w, infoClass, info, dirPath);
1490 if (resp.size() - start > maxOut) {
1491 resp.resize(before);
1492 full = true;
1493 break;
1494 }
1495 if (added > 0) w.put32(lastEntry, (uint32_t)(entry - lastEntry));
1496 lastEntry = entry;
1497 added++;
1498 o->dirIndex++;
1499 if (single) break;
1500 }
1501
1502 if (added == 0) {
1503 if (full) return STATUS_INFO_LENGTH_MISMATCH;
1504 return first ? STATUS_NO_SUCH_FILE : STATUS_NO_MORE_FILES;
1505 }
1506 w.put32(SMB2_HEADER_SIZE + 4, (uint32_t)(resp.size() - start));
1507 return STATUS_SUCCESS;
1508 }

1560 uint32_t onQueryInfo(Connection& c, Request& req) {
1561 uint8_t infoType = req.r.u8(66);
1562 uint8_t infoClass = req.r.u8(67);
1563 uint32_t maxOut = req.r.u32(68);
1564 Open* o = findOpen(c, req, 88);
1565 if (o == nullptr) return fileStatus(req);
1566 FileInfo info;
1567 if (!infoOf(req, *o, info)) return STATUS_OBJECT_NAME_NOT_FOUND;
1568 Share* share = shareOf(req);
1569
1570 Writer w(resp);
1571 w.u16(9);
1572 w.u16(SMB2_HEADER_SIZE + 8);
1573 w.u32(0);
1574 const size_t start = resp.size();
1575 bool variable = false;
1576 uint32_t status = STATUS_SUCCESS;
1577
1578 if (infoType == SMB2_0_INFO_FILE) {
1579 std::string name = windowsPath(o->path);
1580 switch (infoClass) {
1581 case FileBasicInformation:
1582 writeBasicInfo(w, info);
1583 break;
1584 case FileStandardInformation:
1585 writeStandardInfo(w, info, o->deleteOnClose);
1586 break;
1587 case FileInternalInformation:
1588 w.u64(fileIndex(o->path));
1589 break;
1590 case FileEaInformation:
1591 case FileModeInformation:
1592 case FileAlignmentInformation:
1593 w.u32(0);
1594 break;
1595 case FileFullEaInformation:
1596 // no extended attributes: an empty FILE_FULL_EA_INFORMATION list
1597 // is represented by a response with zero length (no entries)
1598 break;
1599 case FileAccessInformation:
1600 w.u32(0x001F01FF);
1601 break;
1602 case FilePositionInformation:
1603 w.u64(0);
1604 break;
1605 case FileAllInformation: {
1606 writeBasicInfo(w, info);
1607 writeStandardInfo(w, info, o->deleteOnClose);
1608 w.u64(fileIndex(o->path));
1609 w.u32(0); // EA
1610 w.u32(0x001F01FF); // access
1611 w.u64(0); // position
1612 w.u32(0); // mode
1613 w.u32(0); // alignment
1614 size_t lenPos = w.pos();
1615 w.u32(0);
1616 w.put32(lenPos, (uint32_t)w.utf16(name));
1617 variable = true;
1618 break;
1619 }
1620 case FileNetworkOpenInformation: {
1621 uint64_t t = toFileTime(info.modified);
1622 for (int i = 0; i < 4; i++) w.u64(t);
1623 w.u64(allocationSize(info.size));
1624 w.u64(info.size);
1625 w.u32(attributes(info));
1626 w.u32(0);
1627 break;
1628 }
1629 case FileAttributeTagInformation:
1630 w.u32(attributes(info));
1631 w.u32(0);
1632 break;
1633 case FileStreamInformation:
1634 if (!info.isDirectory) {
1635 w.u32(0);
1636 size_t lenPos = w.pos();
1637 w.u32(0);
1638 w.u64(info.size);
1639 w.u64(allocationSize(info.size));
1640 w.put32(lenPos, (uint32_t)w.utf16("::$DATA"));
1641 }
1642 variable = true;
1643 break;
1644 case FileCompressionInformation:
1645 w.u64(info.size);
1646 w.u16(0);
1647 w.u8(0);
1648 w.u8(0);
1649 w.u8(0);
1650 w.zeros(3);
1651 break;
1652 case FileNormalizedNameInformation: {
1653 size_t lenPos = w.pos();
1654 w.u32(0);
1655 std::string n = name.size() > 1 ? name.substr(1) : "";
1656 w.put32(lenPos, (uint32_t)w.utf16(n));
1657 variable = true;
1658 break;
1659 }
1660 default:
1661 return STATUS_INVALID_INFO_CLASS;
1662 }
1663 } else if (infoType == SMB2_0_INFO_FILESYSTEM) {
1664 uint64_t total = 0, free = 0;
1665 if (share == nullptr || !share->fs->space(total, free)) {
1666 total = 1ULL << 30;
1667 free = 1ULL << 29;
1668 }
1669 const uint32_t clusterSize = 4096;
1670 switch (infoClass) {
1671 case FileFsVolumeInformation: {
1672 w.u64(startTime);
1673 w.u32((uint32_t)fileIndex(share ? share->name : "IPC$"));
1674 size_t lenPos = w.pos();
1675 w.u32(0);
1676 w.u8(0);
1677 w.u8(0);
1678 w.put32(lenPos, (uint32_t)w.utf16(share ? share->name : "IPC$"));
1679 variable = true;
1680 break;
1681 }
1682 case FileFsSizeInformation:
1683 w.u64(total / clusterSize);
1684 w.u64(free / clusterSize);
1685 w.u32(clusterSize / 512);
1686 w.u32(512);
1687 break;
1688 case FileFsFullSizeInformation:
1689 w.u64(total / clusterSize);
1690 w.u64(free / clusterSize);
1691 w.u64(free / clusterSize);
1692 w.u32(clusterSize / 512);
1693 w.u32(512);
1694 break;
1695 case FileFsDeviceInformation:
1696 w.u32(0x00000007); // FILE_DEVICE_DISK
1697 w.u32(0x00000020); // FILE_DEVICE_IS_MOUNTED
1698 break;
1699 case FileFsAttributeInformation: {
1700 bool cs = share != nullptr && share->fs->isCaseSensitive();
1701 // case preserved names + unicode on disk (+ case sensitive search)
1702 w.u32(0x00000006 | (cs ? 1 : 0));
1703 w.u32(255);
1704 size_t lenPos = w.pos();
1705 w.u32(0);
1706 w.put32(lenPos,
1707 (uint32_t)w.utf16(share ? share->fs->name() : "NTFS"));
1708 variable = true;
1709 break;
1710 }
1711 case FileFsSectorSizeInformation:
1712 for (int i = 0; i < 4; i++) w.u32(512);
1713 w.u32(0x0000000B); // aligned device, partition aligned...
1714 w.u32(0);
1715 w.u32(0);
1716 break;
1717 case FileFsObjectIdInformation:
1718 w.zeros(64);
1719 break;
1720 default:
1721 return STATUS_INVALID_INFO_CLASS;
1722 }
1723 } else if (infoType == SMB2_0_INFO_SECURITY) {
1724 writeSecurityDescriptor(w, info.isDirectory);
1725 if (resp.size() - start > maxOut) {
1726 // the client needs to retry with a bigger buffer
1727 uint32_t needed = (uint32_t)(resp.size() - start);
1728 resp.resize(SMB2_HEADER_SIZE);
1729 Writer e(resp);
1730 e.u16(9);
1731 e.u16(0);
1732 e.u32(4);
1733 e.u32(needed);
1734 keepErrorBody = true;
1735 return STATUS_BUFFER_TOO_SMALL;
1736 }
1737 } else {
1738 return STATUS_NOT_SUPPORTED;
1739 }
1740
1741 size_t len = resp.size() - start;
1742 if (len > maxOut) {
1743 if (!variable) {
1744 return STATUS_INFO_LENGTH_MISMATCH;
1745 }
1746 resp.resize(start + maxOut);
1747 len = maxOut;
1748 status = STATUS_BUFFER_OVERFLOW;
1749 }
1750 w.put32(SMB2_HEADER_SIZE + 4, (uint32_t)len);
1751 return status;
1752 }

1756 uint32_t onSetInfo(Connection& c, Request& req) {
1757 uint8_t infoType = req.r.u8(66);
1758 uint8_t infoClass = req.r.u8(67);
1759 uint32_t len = req.r.u32(68);
1760 uint16_t off = req.r.u16(72);
1761 Open* o = findOpen(c, req, 80);
1762 if (o == nullptr) return fileStatus(req);

1764 if (data == nullptr) return STATUS_INVALID_PARAMETER;
1765 Reader in(data, len);
1766 Share* share = shareOf(req);
1767 if (infoType == SMB2_0_INFO_SECURITY) return body16(2);
1768 if (infoType != SMB2_0_INFO_FILE || share == nullptr)

1771 FileSystem* fs = share->fs;
1772
1773 switch (infoClass) {
1774 case FileBasicInformation:
1775 case FileAllocationInformation:
1776 break; // timestamps and preallocation are ignored
1777 case FileEndOfFileInformation:
1778 if (req.tree->readOnly) return STATUS_MEDIA_WRITE_PROTECTED;
1779 if (o->isDir) return STATUS_INVALID_PARAMETER;
1780 if (!fs->truncate(path.c_str(), in.u64(0))) return STATUS_DISK_FULL;
1781 break;
1782 case FileDispositionInformation:
1783 case FileDispositionInformationEx: {
1784 bool del = infoClass == FileDispositionInformation
1785 ? in.u8(0) != 0
1786 : (in.u32(0) & 1) != 0;
1787 if (del) {
1788 if (req.tree->readOnly || o->path == "/")
1789 return STATUS_ACCESS_DENIED;
1790 if (o->isDir && !isEmptyDir(*fs, path))
1791 return STATUS_DIRECTORY_NOT_EMPTY;
1792 }
1793 o->deleteOnClose = del;
1794 break;
1795 }
1796 case FileRenameInformation: {
1797 if (req.tree->readOnly) return STATUS_MEDIA_WRITE_PROTECTED;
1798 bool replace = in.u8(0) != 0;
1799 uint32_t nameLen = in.u32(16);
1800 std::string target;
1801 if (!toPath(in.utf16(20, nameLen), target) || in.hasError())
1802 return STATUS_OBJECT_NAME_INVALID;
1803 if (target == "/") return STATUS_ACCESS_DENIED;

1805 FileInfo existing;
1806 if (fs->stat(to.c_str(), existing)) {
1807 bool sameFile = equalsIgnoreCase(target, o->path);
1808 if (!sameFile) {
1809 if (!replace || existing.isDirectory)
1810 return STATUS_OBJECT_NAME_COLLISION;
1811 fs->close(to.c_str());
1812 if (!fs->remove(to.c_str())) return STATUS_ACCESS_DENIED;
1813 }
1814 }
1815 if (!parentExists(*share, target)) return STATUS_OBJECT_PATH_NOT_FOUND;
1816 fs->close(path.c_str());
1817 if (!fs->rename(path.c_str(), to.c_str())) return STATUS_ACCESS_DENIED;
1818 // update all opens which refer to the old path
1819 std::string old = o->path;
1820 for (auto& other : c.opens) {
1821 if (other.treeId != o->treeId) continue;
1822 if (other.path == old) {
1823 other.path = target;
1824 } else if (other.path.rfind(old + "/", 0) == 0) {
1825 other.path = target + other.path.substr(old.size());
1826 }
1827 }
1828 SMB_LOGI("rename %s -> %s", path.c_str(), to.c_str());
1829 break;
1830 }
1831 default:
1832 return STATUS_INVALID_INFO_CLASS;
1833 }
1834 return body16(2);
1835 }

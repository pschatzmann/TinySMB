188 SMBFile open(const char* path, const char* mode = "r") {
189 SMBFile file;
190 if (!ensureConnected()) return file;
191 std::string m = mode ? mode : "r";
192 bool plus = m.find('+') != std::string::npos;
193 uint32_t disposition = FILE_OPEN;
194 bool writable = plus;
195 if (m[0] == 'w') {
196 disposition = FILE_OVERWRITE_IF;
197 writable = true;
198 } else if (m[0] == 'a') {
199 disposition = FILE_OPEN_IF;
200 writable = true;
201 }
202 uint32_t access = writable ? ACCESS_READ_WRITE : ACCESS_READ;
203 FileId id;
204 FileInfo info;

206 // directories can not be opened for writing

209 return file;
210 }
211 auto h = std::make_shared<SMBFile::Handle>();
212 h->client = this;
213 h->generation = generation;
214 h->id = id;
215 h->open = true;
216 h->path = normalize(path);
217 h->name = info.name;
218 h->isDir = info.isDirectory;
219 h->writable = writable;
220 h->append = m[0] == 'a';
221 h->size = info.size;
222 h->modified = info.modified;
223 h->pos = h->append ? h->size : 0;
224 file.h = h;

326 uint32_t options, FileId& id, FileInfo& info) {
327 std::string name = toSmbName(path);
328 startRequest(SMB2_CREATE);
329 Writer w(req);
330 w.u16(57);
331 w.u8(0);
332 w.u8(0); // no oplock
333 w.u32(2); // impersonation
334 w.u64(0);
335 w.u64(0);
336 w.u32(access);
337 w.u32(0); // attributes
338 w.u32(7); // share read, write, delete
339 w.u32(disposition);
340 w.u32(options);
341 size_t nameOffPos = w.pos();
342 w.u16(0);
343 size_t nameLenPos = w.pos();
344 w.u16(0);
345 w.u32(0); // create contexts
346 w.u32(0);
347 w.put16(nameOffPos, (uint16_t)w.pos());

349 w.put16(nameLenPos, (uint16_t)n);
350 if (n == 0) w.u8(0);

352 if (transact(resp) != STATUS_SUCCESS) return false;
353 Reader r(resp.data(), resp.size());

355 info.size = r.u64(112);

357 id.persistent = r.u64(128);
358 id.volatileId = r.u64(136);

360 info.name = p == "/" ? "/" : p.substr(p.find_last_of('/') + 1);

377 int64_t readFile(const FileId& id, uint64_t offset, uint8_t* buffer,
378 size_t len) {
379 if (len > maxRead) len = maxRead;
380 startRequest(SMB2_READ);
381 Writer w(req);
382 w.u16(49);
383 w.u8(0x50);
384 w.u8(0);
385 w.u32((uint32_t)len);
386 w.u64(offset);
387 w.u64(id.persistent);
388 w.u64(id.volatileId);
389 w.u32(0); // minimum count
390 w.u32(0); // channel
391 w.u32(0); // remaining
392 w.u16(0);
393 w.u16(0);
394 w.u8(0);
395 std::vector<uint8_t>& resp = rx;
396 uint32_t rc = transact(resp);
397 if (rc == STATUS_END_OF_FILE) return 0;
398 if (rc != STATUS_SUCCESS) return -1;
399 Reader r(resp.data(), resp.size());
400 uint32_t n = r.u32(68);
401 const uint8_t* data = r.ptr(r.u8(66), n);
402 if (data == nullptr || n > len) return -1;
403 memcpy(buffer, data, n);
404 return n;
405 }

449 bool queryDirectory(const FileId& id, bool restart,
450 std::vector<FileInfo>& entries) {
451 entries.clear();
452 startRequest(SMB2_QUERY_DIRECTORY);
453 Writer w(req);
454 w.u16(33);
455 w.u8(FileDirectoryInformation);
456 w.u8(restart ? SMB2_RESTART_SCANS : 0);
457 w.u32(0);
458 w.u64(id.persistent);
459 w.u64(id.volatileId);
460 w.u16(SMB2_HEADER_SIZE + 32);
461 w.u16(2);
462 w.u32(maxTransact);
463 w.utf16("*");
464 std::vector<uint8_t>& resp = rx;
465 if (transact(resp) != STATUS_SUCCESS) return false;
466 Reader r(resp.data(), resp.size());
467 size_t off = r.u16(66);
468 size_t end = off + r.u32(68);
469 while (off + 64 <= end && !r.hasError()) {
470 FileInfo info;
471 info.modified = fromFileTime(r.u64(off + 24));
472 info.size = r.u64(off + 40);
473 info.isDirectory = r.u32(off + 56) & FILE_ATTRIBUTE_DIRECTORY;
474 info.name = r.utf16(off + 64, r.u32(off + 60));
475 if (info.name != "." && info.name != "..") entries.push_back(info);
476 uint32_t next = r.u32(off);
477 if (next == 0) break;
478 off += next;
479 }
480 return true;
481 }

573 bool negotiate() {
574 startRequest(SMB2_NEGOTIATE);
575 Writer w(req);
576 w.u16(36);
577 w.u16(2);
578 w.u16(signingRequired ? SMB2_NEGOTIATE_SIGNING_REQUIRED
579 : SMB2_NEGOTIATE_SIGNING_ENABLED);
580 w.u16(0);
581 w.u32(0);
582 for (int i = 0; i < 16; i++) w.u8(Platform::randomByte());
583 w.u64(0);
584 w.u16(SMB2_DIALECT_0202);
585 w.u16(SMB2_DIALECT_0210);
586 std::vector<uint8_t> resp;
587 if (transact(resp) != STATUS_SUCCESS) return false;
588 Reader r(resp.data(), resp.size());
589 uint16_t serverMode = r.u16(66);
590 if (serverMode & SMB2_NEGOTIATE_SIGNING_REQUIRED) signingRequired = true;
591 uint32_t limit = maxIO;
592 if (limit == 0) {
593 limit = 65536;
594#if defined(ESP32)
595 if (!hasPsram()) limit = 32768;
596#endif
597 }
598 maxTransact = min32(r.u32(92), limit);
599 maxRead = min32(r.u32(96), limit);
600 maxWrite = min32(r.u32(100), limit);
601 SMB_LOGD("dialect 0x%04x, max read %u", r.u16(68), (unsigned)maxRead);
602 return !r.hasError() && maxRead > 0 && maxWrite > 0;
603 }

607 bool sessionSetup() {
608 std::vector<uint8_t> token = ntlm.negotiateToken();
609 std::vector<uint8_t> resp;
610 for (int step = 0; step < 2; step++) {
611 startRequest(SMB2_SESSION_SETUP);
612 Writer w(req);
613 w.u16(25);
614 w.u8(0);
615 w.u8(signingRequired ? SMB2_NEGOTIATE_SIGNING_REQUIRED
616 : SMB2_NEGOTIATE_SIGNING_ENABLED);
617 w.u32(0);
618 w.u32(0);
619 w.u16(SMB2_HEADER_SIZE + 24);
620 w.u16((uint16_t)token.size());
621 w.u64(0);
622 w.bytes(token.data(), token.size());
623 uint32_t rc = transact(resp);
624 Reader r(resp.data(), resp.size());
625 if (step == 0) {
626 if (rc != STATUS_MORE_PROCESSING_REQUIRED) return false;
627 sessionId = r.u64(40);
628 const uint8_t* blob = r.ptr(r.u16(68), r.u16(70));
629 if (blob == nullptr ||
630 !ntlm.authenticateToken(blob, r.u16(70), token)) {
631 status = STATUS_LOGON_FAILURE;
632 return false;
633 }
634 } else {
635 if (rc != STATUS_SUCCESS) return false;
636 bool guest = r.u16(66) & SMB2_SESSION_FLAG_IS_GUEST;
637 if (ntlm.hasSessionKey() && !guest) {
638 memcpy(signingKey, ntlm.sessionKey(), 16);
639 signing = signingRequired;
640 } else if (signingRequired) {
641 SMB_LOGE("signing required, but not possible for guests");
642 return false;
643 }
644 }
645 }
646 return true;
647 }

713 uint32_t transact(std::vector<uint8_t>& resp) {
714 uint64_t id = messageId++;
715 Writer w(req);
716 req[0] = 0xFE;
717 req[1] = 'S';
718 req[2] = 'M';
719 req[3] = 'B';
720 w.put16(4, 64);
721 w.put16(6, 1); // credit charge
722 w.put16(12, command);
723 w.put16(14, 32); // credits requested
724 w.put64(24, id);
725 w.put32(36, treeId);
726 w.put64(40, sessionId);
727 if (signing) {
728 w.put32(16, SMB2_FLAGS_SIGNED);
729 uint8_t mac[32];
730 Crypto::hmacSha256(signingKey, 16, req.data(), req.size(), nullptr, 0,
731 mac);
732 memcpy(req.data() + 48, mac, 16);
733 }
734 status = STATUS_INTERNAL_ERROR;
735 // a single write: separate writes are delayed by Nagle + delayed ACK
736 size_t len = req.size();
737 const uint8_t frame[4] = {0, (uint8_t)(len >> 16), (uint8_t)(len >> 8),
738 (uint8_t)len};
739 req.insert(req.begin(), frame, frame + 4);
740 if (!send(req.data(), req.size())) return fail();
741 // wait for the final response (skip interim STATUS_PENDING responses)
742 uint32_t wait = timeout;
743 while (true) {
744 if (!receive(resp, wait)) return fail();
745 Reader r(resp.data(), resp.size());
746 if (resp.size() < SMB2_HEADER_SIZE || r.u32(0) != 0x424D53FE)
747 return fail();
748 if (r.u64(24) != id) continue;
749 status = r.u32(8);
750 if (status == STATUS_PENDING) {
751 // the server answers later: wait longer than usual
752 wait = pendingTimeout;
753 SMB_LOGD("command %u is pending", command);
754 continue;
755 }
756 if (signing && (r.u32(16) & SMB2_FLAGS_SIGNED) &&
757 !verify(resp.data(), resp.size())) {
758 SMB_LOGE("invalid signature");
759 status = STATUS_ACCESS_DENIED;
760 }
761 if (status != STATUS_SUCCESS && status != STATUS_END_OF_FILE &&
762 status != STATUS_NO_MORE_FILES &&
763 status != STATUS_MORE_PROCESSING_REQUIRED) {
764 SMB_LOGD("command %u -> 0x%08x", command, (unsigned)status);
765 }
766 return status;
767 }
768 }
769
770 uint32_t fail() {
771 SMB_LOGE("connection lost");
772 client.stop();
773 connectedFlag = false;
774 status = STATUS_INTERNAL_ERROR;
775 return status;
776 }
777
778 bool verify(const uint8_t* msg, size_t len) {
779 uint8_t header[SMB2_HEADER_SIZE];
780 memcpy(header, msg, SMB2_HEADER_SIZE);
781 memset(header + 48, 0, 16);
782 uint8_t mac[32];
783 Crypto::hmacSha256(signingKey, 16, header, SMB2_HEADER_SIZE,
784 msg + SMB2_HEADER_SIZE, len - SMB2_HEADER_SIZE, mac);
785 return Crypto::equals(mac, msg + 48, 16);
786 }
787
788 bool send(const uint8_t* data, size_t len) {
789 size_t sent = 0;
790 uint32_t start = millis();
791 while (sent < len) {
792 size_t n = client.write(data + sent, len - sent);
793 if (n == 0 || n > len - sent) {
794 if (!client.connected() || millis() - start > timeout) return false;
795 delay(1);
796 continue;
797 }
798 sent += n;
799 }
800 return true;
801 }
802
803 bool readExact(uint8_t* data, size_t len, uint32_t timeout) {
804 size_t got = 0;
805 uint32_t start = millis();
806 while (got < len) {
807 int avail = client.available();
808 if (avail > 0) {
809 size_t n = len - got < (size_t)avail ? len - got : (size_t)avail;
810 int rc = client.read(data + got, n);
811 if (rc > 0) {
812 got += rc;
813 start = millis();
814 continue;
815 }
816 }
817 if (!client.connected() || millis() - start > timeout) return false;
818 delay(1);
819 }
820 return true;
821 }
822
823 bool receive(std::vector<uint8_t>& resp, uint32_t timeout) {
824 uint8_t frame[4];
825 while (true) {
826 if (!readExact(frame, 4, timeout)) return false;
827 size_t len = ((size_t)frame[1] << 16) | ((size_t)frame[2] << 8) |
828 frame[3];
829 if (frame[0] == 0x85 && len == 0) continue; // keep alive
830 if (frame[0] != 0 || len > (maxRead > maxTransact ? maxRead : maxTransact) + 4096)
831 return false;
832 resp.resize(len);
833 return readExact(resp.data(), len, timeout);
834 }
835 }
836};

881inline size_t SMBFile::read(uint8_t* buffer, size_t len) {
882 if (!isOpen() || h->isDir) return 0;
883 if (!h->wbuf.empty()) h->flushWrite();
884 size_t done = 0;
885 while (done < len) {
886 uint64_t pos = h->pos;
887 // from the read buffer
888 if (pos >= h->rbufOffset && pos < h->rbufOffset + h->rbuf.size()) {
889 size_t off = pos - h->rbufOffset;
890 size_t n = h->rbuf.size() - off;
891 if (n > len - done) n = len - done;
892 memcpy(buffer + done, h->rbuf.data() + off, n);
893 done += n;
894 h->pos += n;
895 continue;
896 }
897 int64_t n;
898 if (len - done >= h->bufferSize) {
899 // big reads go directly to the user buffer
900 n = h->client->readFile(h->id, pos, buffer + done, len - done);
901 if (n > 0) {
902 done += n;
903 h->pos += n;
904 }
905 } else {
906 h->rbuf.resize(h->bufferSize);
907 n = h->client->readFile(h->id, pos, h->rbuf.data(), h->bufferSize);
908 h->rbuf.resize(n > 0 ? n : 0);
909 h->rbufOffset = pos;
910 }
911 if (n <= 0) break;
912 }
913 return done;
914}
915
916inline size_t SMBFile::write(const uint8_t* buffer, size_t len) {
917 if (!isOpen() || !h->writable || h->isDir) return 0;
918 h->rbuf.clear();
919 if (h->append) h->pos = h->size;
920 if (!h->wbuf.empty() && h->pos != h->wbufOffset + h->wbuf.size()) {
921 if (!h->flushWrite()) return 0;
922 }
923 size_t done = 0;
924 if (h->wbuf.empty() && len >= h->bufferSize) {
925 // big writes are sent directly
926 while (done < len) {
927 int64_t n = h->client->writeFile(h->id, h->pos, buffer + done,
928 len - done);
929 if (n <= 0) break;
930 done += n;
931 h->pos += n;
932 }
933 } else {
934 if (h->wbuf.empty()) h->wbufOffset = h->pos;
935 h->wbuf.insert(h->wbuf.end(), buffer, buffer + len);
936 h->pos += len;
937 done = len;
938 if (h->wbuf.size() >= h->bufferSize && !h->flushWrite()) return 0;
939 }
940 if (h->pos > h->size) h->size = h->pos;
941 return done;
942}

944inline bool SMBFile::truncate(uint64_t size) {
945 if (!isOpen() || !h->writable) return false;
946 if (!h->wbuf.empty()) h->flushWrite();
947 h->rbuf.clear();
948 if (!h->client->setFileSize(h->id, size)) return false;
949 h->size = size;
950 if (h->pos > size) h->pos = size;
951 return true;
952}

954inline bool SMBFile::getNextEntry(FileInfo& info) {
955 if (!isOpen() || !h->isDir) return false;
956 while (h->entryIdx >= h->entries.size()) {
957 if (h->dirEnd) return false;
958 bool restart = !h->dirStarted;
959 h->dirStarted = true;
960 h->entryIdx = 0;
961 if (!h->client->queryDirectory(h->id, restart, h->entries)) {
962 h->entries.clear();
963 h->dirEnd = true;
964 return false;
965 }
966 }
967 info = h->entries[h->entryIdx++];
968 return true;
969}

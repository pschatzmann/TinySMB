67 AuthResult process(const uint8_t* blob, size_t len,
68 const std::vector<User>& users, bool allowGuest,
69 std::vector<uint8_t>& response) {
70 response.clear();
71 size_t pos = findNtlmssp(blob, len);
72 if (pos == (size_t)-1) return AuthResult::Failed;
73 if (pos > 0) useSpnego = true;
74 Reader msg(blob + pos, len - pos);
75 uint32_t type = msg.u32(8);
76 if (type == 1) {
77 clientFlags = msg.u32(12);
78 std::vector<uint8_t> challenge = buildChallenge();
79 response = useSpnego ? spnegoResponse(1, &challenge) : challenge;
80 return AuthResult::Continue;
81 }
82 if (type != 3) return AuthResult::Failed;
83 AuthResult result = verify(msg, users, allowGuest);
84 if (result != AuthResult::Failed && useSpnego) {
85 response = spnegoResponse(0, nullptr);
86 }
87 return result;
88 }

108 static size_t findNtlmssp(const uint8_t* p, size_t len) {
109 static const char sig[] = "NTLMSSP";
110 for (size_t i = 0; i + 12 <= len; i++) {
111 if (memcmp(p + i, sig, 8) == 0) return i;
112 }
113 return (size_t)-1;
114 }

117 static std::vector<uint8_t> der(uint8_t tag,
118 const std::vector<uint8_t>& content) {
119 std::vector<uint8_t> out;
120 out.push_back(tag);
121 size_t n = content.size();
122 if (n < 0x80) {
123 out.push_back((uint8_t)n);
124 } else if (n < 0x100) {
125 out.push_back(0x81);
126 out.push_back((uint8_t)n);
127 } else {
128 out.push_back(0x82);
129 out.push_back((uint8_t)(n >> 8));
130 out.push_back((uint8_t)n);
131 }
132 out.insert(out.end(), content.begin(), content.end());
133 return out;
134 }

137 static std::vector<uint8_t> spnegoResponse(uint8_t state,
138 const std::vector<uint8_t>* tok) {
139 std::vector<uint8_t> seq = der(0xA0, {0x0A, 0x01, state});
140 if (tok != nullptr) {
141 std::vector<uint8_t> mech = der(0xA1, ntlmOid());
142 std::vector<uint8_t> token = der(0xA2, der(0x04, *tok));
143 seq.insert(seq.end(), mech.begin(), mech.end());
144 seq.insert(seq.end(), token.begin(), token.end());
145 }
146 return der(0xA1, der(0x30, seq));
147 }

157 std::vector<uint8_t> buildChallenge() {
158 for (int i = 0; i < 8; i++) serverChallenge[i] = randomByte();
159 flags = NEGOTIATE_UNICODE | REQUEST_TARGET | NEGOTIATE_NTLM |
160 NEGOTIATE_ALWAYS_SIGN | TARGET_TYPE_SERVER |
161 NEGOTIATE_EXTENDED_SESSIONSECURITY | NEGOTIATE_TARGET_INFO |
162 NEGOTIATE_VERSION | NEGOTIATE_128 | NEGOTIATE_56;
163 flags |= clientFlags & (NEGOTIATE_SIGN | NEGOTIATE_SEAL |
164 NEGOTIATE_KEY_EXCH | NEGOTIATE_ANONYMOUS);
165
166 std::vector<uint8_t> out;
167 Writer w(out);
168 w.bytes("NTLMSSP", 8);
169 w.u32(2);
170 w.zeros(8); // target name fields: patched below
171 w.u32(flags);
172 w.bytes(serverChallenge, 8);
173 w.zeros(8);
174 w.zeros(8); // target info fields: patched below
175 // version: 6.1 build 7601, NTLM revision 15
176 const uint8_t version[8] = {6, 1, 0xB1, 0x1D, 0, 0, 0, 15};
177 w.bytes(version, 8);
178
179 size_t nameOff = w.pos();
180 size_t nameLen = w.utf16(serverName);
181 w.put16(12, (uint16_t)nameLen);
182 w.put16(14, (uint16_t)nameLen);
183 w.put32(16, (uint32_t)nameOff);
184
185 size_t infoOff = w.pos();
186 avPair(w, 2, domainName); // MsvAvNbDomainName
187 avPair(w, 1, serverName); // MsvAvNbComputerName
188 avPair(w, 4, domainName); // MsvAvDnsDomainName
189 avPair(w, 3, serverName); // MsvAvDnsComputerName
190 w.u32(0); // MsvAvEOL
191 size_t infoLen = w.pos() - infoOff;
192 w.put16(40, (uint16_t)infoLen);
193 w.put16(42, (uint16_t)infoLen);
194 w.put32(44, (uint32_t)infoOff);
195 return out;
196 }

204 AuthResult verify(const Reader& msg, const std::vector<User>& users,
205 bool allowGuest) {
206 auto field = [&](size_t off, size_t& o, size_t& l) {
207 l = msg.u16(off);
208 o = msg.u32(off + 4);
209 return msg.ptr(o, l);
210 };
211 size_t ntOff, ntLen, domOff, domLen, userOff, userLen, keyOff, keyLen;
212 const uint8_t* nt = field(20, ntOff, ntLen);
213 field(28, domOff, domLen);
214 field(36, userOff, userLen);
215 const uint8_t* encKey = field(52, keyOff, keyLen);
216 uint32_t authFlags = msg.u32(60);
217 if (msg.hasError()) return AuthResult::Failed;
218
219 user = msg.utf16(userOff, userLen);
220 std::string domain = msg.utf16(domOff, domLen);
221
222 // anonymous login
223 if (user.empty() && ntLen == 0) {
224 return allowGuest ? AuthResult::Guest : AuthResult::Failed;
225 }
226
227 const User* account = nullptr;
228 for (auto& u : users) {
229 if (upper(u.name) == upper(user)) account = &u;
230 }
231 if (account == nullptr) {
232 return allowGuest ? AuthResult::Guest : AuthResult::Failed;
233 }
234 // we only support NTLMv2 (response = 16 byte proof + blob)
235 if (nt == nullptr || ntLen <= 24) return AuthResult::Failed;
236
237 // NTOWFv2 = HMAC_MD5(MD4(UNICODE(password)), UNICODE(UPPER(user) + dom))
238 std::vector<uint8_t> tmp;
239 Writer w(tmp);
240 w.utf16(account->password);
241 uint8_t ntHash[16];
242 Crypto::md4(tmp.data(), tmp.size(), ntHash);
243 tmp.clear();
244 w.utf16(upper(user));
245 w.utf16(domain);
246 uint8_t responseKey[16];
247 Crypto::hmacMd5(ntHash, 16, tmp.data(), tmp.size(), responseKey);
248
249 // NTProofStr = HMAC_MD5(NTOWFv2, ServerChallenge + blob)
250 tmp.clear();
251 w.bytes(serverChallenge, 8);
252 w.bytes(nt + 16, ntLen - 16);
253 uint8_t proof[16];
254 Crypto::hmacMd5(responseKey, 16, tmp.data(), tmp.size(), proof);
255 if (!Crypto::equals(proof, nt, 16)) return AuthResult::Failed;
256
257 // session key
258 uint8_t baseKey[16];
259 Crypto::hmacMd5(responseKey, 16, proof, 16, baseKey);
260 memcpy(exportedKey, baseKey, 16);
261 if ((authFlags & NEGOTIATE_KEY_EXCH) && encKey != nullptr &&
262 keyLen == 16) {
263 memcpy(exportedKey, encKey, 16);
264 Crypto::rc4(baseKey, 16, exportedKey, 16);
265 }
266 user = account->name;
267 readOnlyUser = account->readOnly;
268 return AuthResult::Success;
269 }

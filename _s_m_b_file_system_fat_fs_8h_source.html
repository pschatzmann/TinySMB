70 int64_t write(const char* path, uint64_t offset, const uint8_t* buffer,
71 size_t len) override {
72 fatfs::FIL* f = openFile(path, true);
73 if (f == nullptr) return -1;
74 uint64_t size = fs.f_size(f);
75 if (offset > size) {
76 // FatFs leaves the gap undefined: fill it with zeros
77 if (!seek(size)) return -1;
78 uint8_t zeros[512] = {0};
79 for (uint64_t pos = size; pos < offset;) {
80 size_t n = offset - pos > sizeof(zeros) ? sizeof(zeros)
81 : (size_t)(offset - pos);
82 if (writeAll(zeros, n) != (int64_t)n) return -1;
83 pos += n;
84 }
85 } else if (!seek(offset)) {
86 return -1;
87 }
88 return writeAll(buffer, len);

199 static void toInfo(const fatfs::FILINFO& fi, FileInfo& info) {
200 info.isDirectory = (fi.fattrib & AM_DIR) != 0;
201 info.size = info.isDirectory ? 0 : (uint64_t)fi.fsize;

206 static uint64_t toUnix(uint16_t date, uint16_t time) {
207 if (date == 0) return 0;
208 int y = 1980 + (date >> 9);
209 unsigned m = (date >> 5) & 0x0F;
210 unsigned d = date & 0x1F;
211 if (m < 1 || m > 12 || d < 1) return 0;
212 // days since 1970-01-01 (Howard Hinnant's days_from_civil)
213 y -= m <= 2;
214 int era = y / 400;
215 unsigned yoe = (unsigned)(y - era * 400);
216 unsigned doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
217 unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
218 int64_t days = (int64_t)era * 146097 + doe - 719468;
219 return (uint64_t)(days * 86400 + (time >> 11) * 3600 +
220 ((time >> 5) & 0x3F) * 60 + (time & 0x1F) * 2);
221 }

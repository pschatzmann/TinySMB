86 static bool bindAck(const Reader& r, uint32_t callId,
87 std::vector<uint8_t>& out) {
88 static const uint8_t NDR[16] = {0x04, 0x5d, 0x88, 0x8a, 0xeb, 0x1c,
89 0xc9, 0x11, 0x9f, 0xe8, 0x08, 0x00,
90 0x2b, 0x10, 0x48, 0x60};
91 uint16_t maxXmit = r.u16(16);
92 uint16_t maxRecv = r.u16(18);
93 uint8_t count = r.u8(24);
94 Writer w(out);
95 header(w, 12, callId);
96 w.u16(maxXmit);
97 w.u16(maxRecv);
98 w.u32(0x12345); // association group
99 const char port[] = "\\PIPE\\srvsvc";
100 w.u16(sizeof(port));
101 w.bytes(port, sizeof(port));
102 w.align(4);
103 w.u8(count);
104 w.u8(0);
105 w.u16(0);
106 size_t off = 28;
107 for (int i = 0; i < count; i++) {
108 uint8_t nTransfer = r.u8(off + 2);
109 int accepted = -1;
110 for (int t = 0; t < nTransfer; t++) {
111 const uint8_t* uuid = r.ptr(off + 24 + t * 20, 16);
112 if (uuid && memcmp(uuid, NDR, 16) == 0) accepted = t;
113 }
114 if (accepted >= 0) {
115 w.u16(0); // acceptance
116 w.u16(0);
117 w.bytes(r.ptr(off + 24 + accepted * 20, 20), 20);
118 } else {
119 w.u16(2); // provider rejection
120 w.u16(2); // proposed transfer syntaxes not supported
121 w.zeros(20);
122 }
123 off += 24 + nTransfer * 20;
124 }
125 if (r.hasError()) return false;
126 w.put16(8, (uint16_t)out.size());
127 return true;
128 }

177 static bool netShareEnumAll(const Reader& in,
178 const std::vector<ShareEntry>& shares,
179 Writer<>& w) {
180 size_t off = skipUniqueString(in, 0);
181 uint32_t level = in.u32(off);
182 if (in.hasError()) return false;
183 uint32_t ref = 0x20000;
184 w.u32(level);
185 w.u32(level);
186 if (level != 0 && level != 1) {
187 w.u32(0); // null container
188 w.u32(0); // total entries
189 w.u32(0); // resume handle
190 w.u32(124); // WERR_INVALID_LEVEL
191 return true;
192 }
193 uint32_t n = (uint32_t)shares.size();
194 w.u32(ref++); // container pointer
195 w.u32(n); // entries read
196 w.u32(ref++); // array pointer
197 w.u32(n); // array max count
198 for (auto& s : shares) {
199 w.u32(ref++); // name pointer
200 if (level == 1) {
201 w.u32(s.type);
202 w.u32(ref++); // comment pointer
203 }
204 }
205 for (auto& s : shares) {
206 string(w, s.name);
207 if (level == 1) string(w, s.comment);
208 }
209 w.u32(n); // total entries
210 w.u32(ref++); // resume handle pointer
211 w.u32(0); // resume handle
212 w.u32(0); // WERR_OK
213 return true;

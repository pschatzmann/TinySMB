68 bool authenticateToken(const uint8_t* blob, size_t len,
69 std::vector<uint8_t>& out) {
70 size_t pos = find(blob, len, "NTLMSSP", 8);
71 if (pos == (size_t)-1) return false;
72 Reader ch(blob + pos, len - pos);
73 if (ch.u32(8) != 2) return false;
74 uint32_t chLen = challengeLength(ch);
75 challengeMsg.assign(blob + pos, blob + pos + chLen);
76 Reader c(challengeMsg.data(), challengeMsg.size());
77 flags = c.u32(20);
78 const uint8_t* serverChallenge = c.ptr(24, 8);
79 size_t infoLen = c.u16(40);
80 const uint8_t* info = c.ptr(c.u32(44), infoLen);
81 if (serverChallenge == nullptr || c.hasError()) return false;
82 std::vector<uint8_t> targetInfo;
83 if (info != nullptr) targetInfo.assign(info, info + infoLen);
84
85 std::vector<uint8_t> auth;
86 useMic = false;
87 if (isAnonymous()) {
88 buildAnonymous(auth);
89 } else if (!buildAuthenticate(serverChallenge, targetInfo, auth)) {
90 return false;
91 }
92
93 std::vector<uint8_t> seq = der(0xA2, der(0x04, auth));
94 if (useMic) {
95 std::vector<uint8_t> mic = der(0xA3, der(0x04, mechListMic()));
96 seq.insert(seq.end(), mic.begin(), mic.end());
97 }
98 out = der(0xA1, der(0x30, seq));
99 return true;
100 }

128 static std::vector<uint8_t> der(uint8_t tag,
129 const std::vector<uint8_t>& content) {
130 std::vector<uint8_t> out;
131 out.push_back(tag);
132 size_t n = content.size();
133 if (n < 0x80) {
134 out.push_back((uint8_t)n);
135 } else if (n < 0x100) {
136 out.push_back(0x81);
137 out.push_back((uint8_t)n);
138 } else {
139 out.push_back(0x82);
140 out.push_back((uint8_t)(n >> 8));
141 out.push_back((uint8_t)n);
142 }
143 out.insert(out.end(), content.begin(), content.end());
144 return out;
145 }

155 static uint32_t challengeLength(const Reader& r) {
156 uint32_t end = 56;

190 bool buildAuthenticate(const uint8_t* serverChallenge,
191 std::vector<uint8_t> info, std::vector<uint8_t>& auth) {
192 // NTOWFv2
193 std::vector<uint8_t> tmp;
194 Writer t(tmp);
195 t.utf16(password);
196 uint8_t ntHash[16];
197 Crypto::md4(tmp.data(), tmp.size(), ntHash);
198 tmp.clear();
199 t.utf16(upper(user));
200 t.utf16(domain);
201 uint8_t responseKey[16];
202 Crypto::hmacMd5(ntHash, 16, tmp.data(), tmp.size(), responseKey);
203
204 // timestamp from the server: then we must provide a MIC
205 uint64_t timestamp = 0;
206 size_t eol = (size_t)-1;
207 size_t flagsPos = (size_t)-1;
208 Reader r(info.data(), info.size());
209 for (size_t off = 0; off + 4 <= info.size();) {
210 uint16_t id = r.u16(off);
211 uint16_t len = r.u16(off + 2);
212 if (id == 0) {
213 eol = off;
214 break;
215 }
216 if (id == 7 && len == 8) timestamp = r.u64(off + 4);
217 if (id == 6 && len == 4) flagsPos = off + 4;
218 off += 4 + len;
219 }
220 if (eol == (size_t)-1) {
221 eol = info.size();
222 info.insert(info.end(), 4, 0);
223 }
224 useMic = timestamp != 0;
225 SMB_LOGD("NTLM MIC: %s", useMic ? "yes" : "no");
226 if (useMic) {
227 Writer iw(info);
228 if (flagsPos != (size_t)-1) {
229 iw.put32(flagsPos, r.u32(flagsPos) | 0x2); // MIC provided
230 } else {
231 const uint8_t avFlags[8] = {6, 0, 4, 0, 2, 0, 0, 0};
232 info.insert(info.begin() + eol, avFlags, avFlags + 8);
233 }
234 } else {
235 timestamp = toFileTime(Platform::unixTime());
236 }
237
238 uint8_t clientChallenge[8];
239 for (auto& b : clientChallenge) b = Platform::randomByte();
240
241 // NTLMv2 response = NTProofStr + temp
242 std::vector<uint8_t> temp;
243 Writer tw(temp);
244 tw.u8(1);
245 tw.u8(1);
246 tw.zeros(6);
247 tw.u64(timestamp);
248 tw.bytes(clientChallenge, 8);
249 tw.u32(0);
250 tw.bytes(info.data(), info.size());
251 tw.u32(0);
252 tmp.assign(serverChallenge, serverChallenge + 8);
253 tmp.insert(tmp.end(), temp.begin(), temp.end());
254 uint8_t proof[16];
255 Crypto::hmacMd5(responseKey, 16, tmp.data(), tmp.size(), proof);
256 std::vector<uint8_t> nt(proof, proof + 16);
257 nt.insert(nt.end(), temp.begin(), temp.end());
258
259 // LMv2 response (zeros if a MIC is used)
260 std::vector<uint8_t> lm(24, 0);
261 if (!useMic) {
262 tmp.assign(serverChallenge, serverChallenge + 8);
263 tmp.insert(tmp.end(), clientChallenge, clientChallenge + 8);
264 Crypto::hmacMd5(responseKey, 16, tmp.data(), tmp.size(), lm.data());
265 memcpy(lm.data() + 16, clientChallenge, 8);
266 }
267
268 // session key
269 uint8_t baseKey[16];
270 Crypto::hmacMd5(responseKey, 16, proof, 16, baseKey);
271 uint8_t encryptedKey[16];
272 bool keyExchange = flags & NEGOTIATE_KEY_EXCH;
273 if (keyExchange) {
274 for (auto& b : exportedKey) b = Platform::randomByte();
275 memcpy(encryptedKey, exportedKey, 16);
276 Crypto::rc4(baseKey, 16, encryptedKey, 16);
277 } else {
278 memcpy(exportedKey, baseKey, 16);
279 }
280
281 // AUTHENTICATE message
282 Writer w(auth);
283 w.bytes("NTLMSSP", 8);
284 w.u32(3);
285 w.zeros(48); // fields: patched below
286 w.u32(flags);
287 writeVersion(w);
288 size_t micPos = w.pos();
289 w.zeros(16);
290 field(w, 12, w.pos(), lm.size());
291 w.bytes(lm.data(), lm.size());
292 field(w, 20, w.pos(), nt.size());
293 w.bytes(nt.data(), nt.size());
294 size_t off = w.pos();
295 field(w, 28, off, w.utf16(domain));
296 off = w.pos();
297 field(w, 36, off, w.utf16(user));
298 off = w.pos();
299 field(w, 44, off, w.utf16("ARDUINO"));
300 field(w, 52, w.pos(), keyExchange ? 16 : 0);
301 if (keyExchange) w.bytes(encryptedKey, 16);
302
303 if (useMic) {
304 tmp = negotiateMsg;
305 tmp.insert(tmp.end(), challengeMsg.begin(), challengeMsg.end());
306 tmp.insert(tmp.end(), auth.begin(), auth.end());
307 Crypto::hmacMd5(exportedKey, 16, tmp.data(), tmp.size(),
308 auth.data() + micPos);
309 }
310 return true;
311 }

314 std::vector<uint8_t> mechListMic() {
315 static const char signMagic[] =
316 "session key to client-to-server signing key magic constant";
317 static const char sealMagic[] =
318 "session key to client-to-server sealing key magic constant";
319 auto deriveKey = [&](const char* magic, size_t len, uint8_t out[16]) {
320 std::vector<uint8_t> d(exportedKey, exportedKey + 16);
321 d.insert(d.end(), (const uint8_t*)magic, (const uint8_t*)magic + len);
322 Crypto::md5(d.data(), d.size(), out);
323 };
324 uint8_t signKey[16], sealKey[16];
325 deriveKey(signMagic, sizeof(signMagic), signKey); // including \0
326 deriveKey(sealMagic, sizeof(sealMagic), sealKey);
327
328 std::vector<uint8_t> data = {0, 0, 0, 0}; // sequence number
329 std::vector<uint8_t> types = mechTypes();
330 data.insert(data.end(), types.begin(), types.end());
331 uint8_t mac[16];
332 Crypto::hmacMd5(signKey, 16, data.data(), data.size(), mac);
333 if (flags & NEGOTIATE_KEY_EXCH) Crypto::rc4(sealKey, 16, mac, 8);
334
335 std::vector<uint8_t> sig;
336 Writer w(sig);
337 w.u32(1);
338 w.bytes(mac, 8);
339 w.u32(0);
340 return sig;
341 }
